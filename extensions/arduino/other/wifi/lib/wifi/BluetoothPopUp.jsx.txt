import React, { useEffect, useMemo, useRef, useState } from "react";

const BluetoothConnectPopup = ({
    open,
    onClose,
    onConnected,
    onDisconnected,
    allowAutoReconnect = false
}) => {
    const [mode, setMode] = useState(null);
    const [devices, setDevices] = useState([]);
    const [loading, setLoading] = useState(false);
    const [connectingId, setConnectingId] = useState(null);
    const [connectedId, setConnectedId] = useState(null);
    const mountedRef = useRef(false);
    const onDisconnectedRef = useRef(onDisconnected);
    const reconnectingRef = useRef(false);

    useEffect(() => {
        mountedRef.current = true;
        return () => {
            mountedRef.current = false;
        };
    }, []);

    useEffect(() => {
        onDisconnectedRef.current = onDisconnected;
    }, [onDisconnected]);

    useEffect(() => {
        window.bluetoothAutoReconnectEnabled = Boolean(allowAutoReconnect);
        if (!allowAutoReconnect) {
            window.bluetoothReconnectPromise = null;
        }
    }, [allowAutoReconnect]);

    // Known service and characteristic UUIDs
    const uuids = useMemo(
        () => ({
            nanoService: "0000ffe0-0000-1000-8000-00805f9b34fb",
            nanoWriteChar: "0000ffe2-0000-1000-8000-00805f9b34fb",
            nanoNotifyChar: "0000ffe1-0000-1000-8000-00805f9b34fb",

            espServiceCustom: "d804b643-6ce7-4e81-9f8a-ce0f699085eb",
            espCharCustom: "c8659211-af91-4ad3-a995-a58d6fd26145",

            nusService: "6e400001-b5a3-f393-e0a9-e50e24dcca9e",
            nusCharRX: "6e400002-b5a3-f393-e0a9-e50e24dcca9e"
        }),
        []
    );

    const handleModeSelect = (type) => {
        setMode(type);
        setDevices([]);
        setConnectedId(null);
    };

    const isGenericDisplayName = name =>
        /^(unknown|unnamed|device|bluetooth device|ai\s*&\s*iot\s*\(nano\)|ai\s*&\s*robotics\s*\(esp32\)|n\/a|\s*)$/i.test(
            String(name || "").trim()
        );

    const normalizeName = name =>
        String(name || "")
            .trim()
            .toLowerCase();

    const tryReadGapDeviceName = async device => {
        if (!device || !device.gatt) {
            return null;
        }
        const wasConnected = Boolean(device.gatt.connected);
        try {
            const server = wasConnected ? device.gatt : await device.gatt.connect();
            const service = await server.getPrimaryService("generic_access");
            const characteristic = await service.getCharacteristic("gap.device_name");
            const value = await characteristic.readValue();
            const gapName = new TextDecoder("utf-8").decode(value).trim();
            if (gapName && !/^(unknown|unnamed|device|n\/a|\s*)$/i.test(gapName)) {
                return gapName;
            }
        } catch (err) {
            console.warn("Could not read GAP device name:", err);
        } finally {
            if (!wasConnected && device.gatt && device.gatt.connected) {
                try {
                    device.gatt.disconnect();
                } catch (e) {
                    // ignore disconnect errors
                }
            }
        }
        return null;
    };

    const makeFallbackName = (device, currentMode) => {
        const raw = String(device && device.id ? device.id : "");
        const idTail = raw.replace(/[^A-Fa-f0-9]/g, "").slice(-6) || "unknown";
        if (currentMode === "nano") {
            return `AI & IOT (Nano) · ${idTail}`;
        }
        if (currentMode === "esp32") {
            return `AI & Robotics (ESP32) · ${idTail}`;
        }
        return `Bluetooth device · ${idTail}`;
    };

    const resolveNanoGatt = async server => {
        console.log("🔵 Resolving Nano GATT services...");

        const service = await server.getPrimaryService(uuids.nanoService);
        const characteristics = await service.getCharacteristics();

        characteristics.forEach((chr, index) => {
            console.log(`Nano characteristic ${index + 1}: ${chr.uuid}`, {
                
                read: chr.properties.read,
                write: chr.properties.write,
                writeWithoutResponse: chr.properties.writeWithoutResponse,
                notify: chr.properties.notify,
                indicate: chr.properties.indicate
            });
        });

        const preferredWrite =
            characteristics.find(chr => chr.uuid === uuids.nanoWriteChar) ||
            characteristics.find(
                chr =>
                    chr.properties.write ||
                    chr.properties.writeWithoutResponse ||
                    chr.properties.reliableWrite
            ) ||
            characteristics.find(chr => chr.uuid === uuids.nanoNotifyChar);

        const preferredNotify =
            characteristics.find(chr => chr.uuid === uuids.nanoNotifyChar) ||
            characteristics.find(chr => chr.properties.notify || chr.properties.indicate) ||
            preferredWrite;

        if (!preferredWrite) {
            throw new Error("No usable Nano characteristic found.");
        }

        const alternateCharacteristic =
            characteristics.find(
                chr =>
                    chr.uuid !== preferredWrite.uuid &&
                    (chr.properties.write ||
                        chr.properties.writeWithoutResponse ||
                        chr.uuid === uuids.nanoNotifyChar)
            ) || null;

        return {
            service,
            characteristic: preferredWrite,
            notifyCharacteristic: preferredNotify,
            alternateCharacteristic
        };
    };

    const clearBluetoothGlobals = deviceToClear => {
        const activeDevice = deviceToClear || window.bluetoothDevice;
        const disconnectHandler = window.bluetoothDisconnectHandler;

        if (activeDevice && disconnectHandler) {
            try {
                activeDevice.removeEventListener(
                    "gattserverdisconnected",
                    disconnectHandler
                );
            } catch (err) {
                console.warn("Could not remove disconnect listener:", err);
            }
        }

        window.bluetoothDevice = null;
        window.bluetoothCharacteristic = null;
        window.bluetoothNotifyChar = null;
        window.bluetoothAlternateCharacteristic = null;
        window.bluetoothDisconnectHandler = null;
        window.bluetoothMode = null;
        window.bluetoothWriteQueue = Promise.resolve();
        window.bluetoothReconnectPromise = null;
        window.bluetoothResponseQueue = [];
        window.bluetoothLastResponse = "";
    };

    const installDisconnectHandler = device => {
        window.bluetoothDisconnectHandler = event => {
            const disconnectedDevice = event?.target || device;
            handleDisconnectedState(disconnectedDevice);
        };

        device.addEventListener(
            "gattserverdisconnected",
            window.bluetoothDisconnectHandler
        );
    };

    const applyResolvedConnection = async ({
        device,
        server,
        mode: currentMode,
        name,
        id,
        isReconnect = false
    }) => {
        let characteristic;
        let notifyCharacteristic = null;
        let alternateCharacteristic = null;

        if (currentMode === "nano") {
            const resolvedNano = await resolveNanoGatt(server);
            characteristic = resolvedNano.characteristic;
            notifyCharacteristic = resolvedNano.notifyCharacteristic;
            alternateCharacteristic = resolvedNano.alternateCharacteristic;

            try {
                if (
                    notifyCharacteristic &&
                    (notifyCharacteristic.properties.notify ||
                        notifyCharacteristic.properties.indicate)
                ) {
                    await notifyCharacteristic.startNotifications();
                    window.bluetoothLastResponse = "";

                    notifyCharacteristic.addEventListener(
                        "characteristicvaluechanged",
                        event => {
                            const text = new TextDecoder().decode(event.target.value);
                            console.log("Nano Data Received:", text);
                            const responseQueue = window.bluetoothResponseQueue || [];
                            text.split(/\r?\n/).map(item => item.trim()).filter(Boolean).forEach(item => {
                                responseQueue.push(item);
                            });
                            window.bluetoothResponseQueue = responseQueue;
                            window.bluetoothLastResponse =
                                (window.bluetoothLastResponse || "") + text;
                        }
                    );

                    console.log("✅ Nano notification listener attached");
                } else {
                    console.warn("⚠️ Nano notify characteristic not available");
                }
            } catch (notifyErr) {
                console.warn("⚠️ Nano notifications not supported:", notifyErr);
            }
        } else {
            console.log("🔵 Resolving ESP32 GATT services...");
            const resolved = await resolveEsp32Gatt(server);
            characteristic = resolved.characteristic;
            console.log("✅ ESP32 characteristic found:", characteristic.uuid);

            try {
                if (characteristic.properties.notify || characteristic.properties.indicate) {
                    await characteristic.startNotifications();
                    console.log("✅ ESP32 Bluetooth notifications started");

                    window.bluetoothResponseQueue = window.bluetoothResponseQueue || [];
                    characteristic.addEventListener(
                        "characteristicvaluechanged",
                        event => {
                            const value = event.target.value;
                            const text = new TextDecoder().decode(value);
                            console.log("📥 ESP32 Data Received:", text);
                            text.split(/\r?\n/).map(item => item.trim()).filter(Boolean).forEach(item => {
                                window.bluetoothResponseQueue.push(item);
                            });
                            window.bluetoothLastResponse =
                                (window.bluetoothLastResponse || "") + text;
                        }
                    );

                    console.log("✅ ESP32 notification listener attached");
                } else {
                    console.warn("⚠️ Characteristic does not support notifications");
                }
            } catch (notifError) {
                console.warn("⚠️ ESP32 Notifications not supported:", notifError);
            }
        }

        clearBluetoothGlobals(window.bluetoothDevice);

        window.bluetoothDevice = device;
        window.bluetoothCharacteristic = characteristic;
        window.bluetoothNotifyChar = notifyCharacteristic;
        window.bluetoothAlternateCharacteristic = alternateCharacteristic;
        window.bluetoothMode = currentMode;
        window.bluetoothManualDisconnect = false;
        window.bluetoothStickySession = true;
        window.bluetoothAutoReconnectEnabled = Boolean(allowAutoReconnect);
        window.bluetoothSession = {device, mode: currentMode, name, id};
        installDisconnectHandler(device);

        if (mountedRef.current) {
            setConnectedId(id || device.id);
            setConnectingId(null);
        }

        if (onConnected) {
            console.log("🔵 Calling onConnected callback...");
            onConnected(device, name, {isReconnect});
        }

        console.log("✅ Successfully connected to:", device.name || name || "Bluetooth Device");
        console.log("✅ Device ID:", device.id);
        console.log("✅ Characteristic UUID:", characteristic.uuid);
        console.log("✅ Characteristic properties:", characteristic.properties);
    };

    const tryAutoReconnect = async disconnectedDevice => {
        if (reconnectingRef.current) return false;
        if (window.bluetoothManualDisconnect) return false;
        if (!window.bluetoothStickySession) return false;
        if (!window.bluetoothAutoReconnectEnabled) return false;

        const session = window.bluetoothSession;
        if (!session || !session.device) return false;
        if (disconnectedDevice && session.device.id !== disconnectedDevice.id) return false;

        reconnectingRef.current = true;
        window.bluetoothCharacteristic = null;
        window.bluetoothNotifyChar = null;
        window.bluetoothAlternateCharacteristic = null;

        const reconnectJob = (async () => {
            let attempt = 0;
            while (
                !window.bluetoothManualDisconnect &&
                window.bluetoothStickySession &&
                window.bluetoothAutoReconnectEnabled
            ) {
                attempt += 1;
                try {
                    const delayMs = Math.min(3000, 500 + (attempt * 250));
                    await new Promise(resolve => setTimeout(resolve, delayMs));

                    const server = await session.device.gatt.connect();
                    await new Promise(resolve => setTimeout(resolve, 500));

                    await applyResolvedConnection({
                        device: session.device,
                        server,
                        mode: session.mode,
                        name: session.name,
                        id: session.id,
                        isReconnect: true
                    });

                    reconnectingRef.current = false;
                    window.bluetoothReconnectPromise = null;
                    return true;
                } catch (err) {
                    // Keep retrying quietly until the user disconnects manually.
                }
            }

            reconnectingRef.current = false;
            window.bluetoothReconnectPromise = null;
            return false;
        })();

        window.bluetoothReconnectPromise = reconnectJob;
        return reconnectJob;
    };

    const handleDisconnectedState = async disconnectedDevice => {
        const activeDevice = window.bluetoothDevice;
        if (activeDevice && disconnectedDevice && activeDevice.id !== disconnectedDevice.id) {
            console.log("Ignoring stale disconnect event from old device:", disconnectedDevice.id);
            return;
        }

        const reconnected = await tryAutoReconnect(disconnectedDevice || activeDevice);
        if (reconnected) {
            return;
        }

        clearBluetoothGlobals(disconnectedDevice || activeDevice);

        if (mountedRef.current) {
            setConnectedId(null);
        }

        if (onDisconnectedRef.current) {
            onDisconnectedRef.current();
        }
    };

    // ============================
    // SCAN DEVICES (NO CONNECT!)
    // ============================
    const addOneDevice = async () => {
        if (!mode) return;
        setLoading(true);

        try {
            const options = {
                acceptAllDevices: true,
                optionalServices: mode === "nano"
                    ? [
                        "generic_access",
                        "battery_service",
                        uuids.nanoService
                    ]
                    : [
                        "generic_access",
                        "battery_service",
                        uuids.nanoService,
                        uuids.espServiceCustom,
                        uuids.nusService,
                        "0000180f-0000-1000-8000-00805f9b34fb", // Battery Service
                        "0000180a-0000-1000-8000-00805f9b34fb"  // Device Information
                    ]
            };

            const device = await navigator.bluetooth.requestDevice(options);
            if (!device) throw new Error("No device selected");

            let name = device.name?.trim() || "";

            // If the device name is missing/generic, try reading GAP name first.
            if (!name || isGenericDisplayName(name)) {
                const gapName = await tryReadGapDeviceName(device);
                if (gapName) {
                    name = gapName;
                    console.log("Resolved firmware name via GAP:", name);
                } else {
                    name = makeFallbackName(device, mode);
                    if (mode === "esp32") console.log("🔵 ESP32 device detected (unnamed)");
                    if (mode === "nano") console.log("🔵 Nano device detected (unnamed)");
                }
            }

            if (name) {
                // Check if name contains ESP32 or similar indicators
                if (name.toLowerCase().includes("esp32") || name.toLowerCase().includes("esp")) {
                    console.log("🔵 ESP32 device detected:", name);
                }
            }

            const isGenericName = /^(unknown|unnamed|device|n\/a|\s*)$/i.test(name);

            if (mode === "esp32" && (name === "AI & Robotics (ESP32)" || isGenericName)) {
                const wasConnected = Boolean(device.gatt?.connected);

                try {
                    const server = wasConnected ? device.gatt : await device.gatt.connect();
                    const service = await server.getPrimaryService("generic_access");
                    const characteristic = await service.getCharacteristic("gap.device_name");
                    const value = await characteristic.readValue();
                    const gapName = new TextDecoder("utf-8").decode(value).trim();

                    if (gapName && !/^(unknown|unnamed|device|n\/a|\s*)$/i.test(gapName)) {
                        name = gapName;
                        console.log("Resolved ESP32 firmware name:", name);
                    }
                } catch (gapError) {
                    console.warn("Could not read ESP32 GAP device name:", gapError);
                } finally {
                    if (!wasConnected && device.gatt?.connected) {
                        device.gatt.disconnect();
                    }
                }
            }

            const id = device.id;

            setDevices(prev => {
                const existingIndex = prev.findIndex(
                    d => d.id === id || normalizeName(d.name) === normalizeName(name)
                );

                if (existingIndex === -1) {
                    return [...prev, { name, id, device }];
                }

                const existing = prev[existingIndex];
                const shouldReplace =
                    existing.id === id ||
                    (isGenericDisplayName(existing.name) && !isGenericDisplayName(name));

                if (!shouldReplace) {
                    return prev;
                }

                return prev.map((item, index) =>
                    index === existingIndex ? { name, id, device } : item
                );
            });
        } catch (err) {
            if (!/cancel/i.test(err.message)) {
                console.error("Bluetooth scan failed:", err);
            }
        }

        setLoading(false);
    };

    // ==================================
    // RESOLVE ESP32 CHARACTERISTIC
    // ==================================
    const resolveEsp32Gatt = async (server) => {
        console.log("🔵 Trying ESP32 custom service...");
        try {
            const svc = await server.getPrimaryService(uuids.espServiceCustom);
            console.log("✅ Found ESP32 custom service");
            const chr = await svc.getCharacteristic(uuids.espCharCustom);
            console.log("✅ Found ESP32 custom characteristic");
            return { service: svc, characteristic: chr };
        } catch (err) {
            console.warn("⚠️ ESP32 custom service not found:", err.message);
        }

        console.log("🔵 Trying NUS service...");
        try {
            const nus = await server.getPrimaryService(uuids.nusService);
            console.log("✅ Found NUS service");
            const rx = await nus.getCharacteristic(uuids.nusCharRX);
            console.log("✅ Found NUS RX characteristic");
            return { service: nus, characteristic: rx };
        } catch (err) {
            console.warn("⚠️ NUS service not found:", err.message);
        }

        console.log("🔵 Trying fallback - searching all services...");
        // fallback — find any writable characteristic
        try {
            // ✅ Try to get services with retry
            let services = [];
            let retries = 3;
            
            while (retries > 0 && services.length === 0) {
                try {
                    services = await server.getPrimaryServices();
                    if (services.length === 0) {
                        console.log(`⚠️ No services found, retrying... (${retries} attempts left)`);
                        await new Promise(resolve => setTimeout(resolve, 500));
                        retries--;
                    }
                } catch (err) {
                    console.warn("⚠️ Error getting services, retrying...", err.message);
                    await new Promise(resolve => setTimeout(resolve, 500));
                    retries--;
                }
            }
            
            if (services.length === 0) {
                throw new Error("No services found on device after retries");
            }
            
            console.log(`✅ Found ${services.length} services, searching for writable characteristic...`);
            
            // Log all services for debugging
            services.forEach((svc, idx) => {
                console.log(`  Service ${idx + 1}: ${svc.uuid}`);
            });
            
            for (let svc of services) {
                try {
                    console.log(`🔍 Checking service: ${svc.uuid}`);
                    const chrs = await svc.getCharacteristics();
                    console.log(`  Found ${chrs.length} characteristics in this service`);
                    
                    // Log all characteristics
                    chrs.forEach((chr, idx) => {
                        console.log(`    Characteristic ${idx + 1}: ${chr.uuid}, properties:`, {
                            read: chr.properties.read,
                            write: chr.properties.write,
                            writeWithoutResponse: chr.properties.writeWithoutResponse,
                            notify: chr.properties.notify
                        });
                    });
                    
                    // First try to find writable characteristic
                    let writable = chrs.find(
                        (c) =>
                            c.properties.write ||
                            c.properties.writeWithoutResponse ||
                            c.properties.reliableWrite
                    );

                    if (writable) {
                        console.log("✅ Found writable characteristic in service:", svc.uuid);
                        return { service: svc, characteristic: writable };
                    }
                    
                    // If no writable, try to find notify characteristic (for receiving data)
                    let notifyChar = chrs.find((c) => c.properties.notify || c.properties.indicate);
                    if (notifyChar) {
                        console.log("⚠️ No writable characteristic, but found notify characteristic:", svc.uuid);
                        // Only use notify if it also has write capability
                        if (notifyChar.properties.write || notifyChar.properties.writeWithoutResponse) {
                            console.log("✅ Notify characteristic also has write capability");
                            return { service: svc, characteristic: notifyChar };
                        } else {
                            console.warn("⚠️ Notify characteristic found but no write capability");
                        }
                    }
                    
                    // Last resort: use first characteristic with any write capability
                    let anyWriteChar = chrs.find((c) => 
                        c.properties.write || 
                        c.properties.writeWithoutResponse || 
                        c.properties.reliableWrite
                    );
                    if (anyWriteChar) {
                        console.log("⚠️ Using characteristic with write capability:", svc.uuid);
                        return { service: svc, characteristic: anyWriteChar };
                    }
                    
                    // If still nothing, warn but don't use read-only characteristics
                    console.warn("⚠️ No writable characteristic found in service:", svc.uuid);
                    
                    console.log(`  ⚠️ No usable characteristic in service: ${svc.uuid}`);
                } catch (err) {
                    console.warn(`⚠️ Error reading characteristics from ${svc.uuid}:`, err.message);
                }
            }
        } catch (err) {
            console.error("❌ Error getting primary services:", err);
            console.error("❌ Full error:", err);
        }

        throw new Error("No valid characteristic found on ESP32. Please check device firmware.");
    };

    // ============================
    // CONNECT DEVICE (REAL CONNECT)
    // ============================
    const connectDevice = async (entry) => {
        const { device: storedDevice, id } = entry;
        setConnectingId(id);

        try {
            // ✅ IMPORTANT: Request fresh device connection
            // Stored device object might be stale, so we request fresh connection
            console.log("🔵 Requesting fresh device connection...");
            
            let device;
            let server;
            
            // Try to use stored device first if it's still valid
            try {
                if (storedDevice && storedDevice.gatt) {
                    // Check if device is already connected
                    if (storedDevice.gatt.connected) {
                        console.log("🔵 Using existing connection...");
                        device = storedDevice;
                        server = storedDevice.gatt;
                    } else {
                        // Try to connect with stored device
                        console.log("🔵 Connecting with stored device object...");
                        server = await storedDevice.gatt.connect();
                        device = storedDevice;
                        console.log("✅ Connected with stored device");
                        
                        // ✅ Wait for services to be ready
                        console.log("⏳ Waiting for services to be ready...");
                        await new Promise(resolve => setTimeout(resolve, 1000));
                        
                        // ✅ Debug: List all available services
                        try {
                            const allServices = await server.getPrimaryServices();
                            console.log(`📋 Found ${allServices.length} total services on device:`);
                            allServices.forEach((svc, idx) => {
                                console.log(`  ${idx + 1}. Service UUID: ${svc.uuid}`);
                            });
                        } catch (debugErr) {
                            console.warn("⚠️ Could not list services:", debugErr);
                        }
                    }
                } else {
                    throw new Error("Stored device invalid");
                }
            } catch (storedErr) {
                // If stored device fails, request fresh device
                console.log("⚠️ Stored device failed, requesting fresh device...");
                console.log("Error:", storedErr);
                
                const options = {
                    acceptAllDevices: true,
                    optionalServices: mode === "nano" 
                        ? [uuids.nanoService, "generic_access", "battery_service"]
                        : [
                            uuids.espServiceCustom, 
                            uuids.nusService,
                            "generic_access",
                            "battery_service",
                            "0000180f-0000-1000-8000-00805f9b34fb", // Battery Service
                            "0000180a-0000-1000-8000-00805f9b34fb"  // Device Information
                        ]
                };
                
                // Request fresh device (browser will remember previous selection)
                device = await navigator.bluetooth.requestDevice(options);
                
                if (!device) {
                    throw new Error("No device selected");
                }
                
                // Connect to fresh device
                console.log("🔵 Connecting to fresh device...");
                server = await device.gatt.connect();
                console.log("✅ Connected to fresh device");
                
                // ✅ IMPORTANT: Wait a bit for services to be ready
                console.log("⏳ Waiting for services to be ready...");
                await new Promise(resolve => setTimeout(resolve, 1000));
                
                // ✅ Debug: List all available services
                try {
                    const allServices = await server.getPrimaryServices();
                    console.log(`📋 Found ${allServices.length} total services on device:`);
                    allServices.forEach((svc, idx) => {
                        console.log(`  ${idx + 1}. Service UUID: ${svc.uuid}`);
                    });
                } catch (debugErr) {
                    console.warn("⚠️ Could not list services:", debugErr);
                }
            }

            await applyResolvedConnection({
                device,
                server,
                mode,
                name: entry.name,
                id: entry.id
            });
        } catch (err) {
            console.error("❌ Connection failed:", err);
            console.error("❌ Error details:", {
                name: err.name,
                message: err.message,
                code: err.code
            });
            
            // Better error message
            let errorMsg = "Connection failed. ";
            if (err.name === "NetworkError") {
                errorMsg += "Device might be out of range or turned off.";
            } else if (err.name === "SecurityError") {
                errorMsg += "Bluetooth permission denied.";
            } else if (err.message && err.message.includes("already connected")) {
                errorMsg += "Device is already connected. Please disconnect first.";
            } else {
                errorMsg += "Please try again. Make sure device is powered on and in range.";
            }
            
            alert(errorMsg);
            if (mountedRef.current) {
                setConnectingId(null);
            }
        }
    };

    // ============================
    // DISCONNECT
    // ============================
    const handleManualDisconnect = async () => {
        try {
            window.bluetoothManualDisconnect = true;
            window.bluetoothStickySession = false;
            const device = window.bluetoothDevice;
            const characteristic = window.bluetoothCharacteristic;
            const notifyChar = window.bluetoothNotifyChar;
            const disconnectHandler = window.bluetoothDisconnectHandler;

            if (device && disconnectHandler) {
                try {
                    device.removeEventListener("gattserverdisconnected", disconnectHandler);
                } catch (listenerErr) {
                    console.warn("Could not detach disconnect listener:", listenerErr);
                }
            }

            if (device?.gatt?.connected) {
                try {
                    // Stop notifications before disconnecting
                    const targetChar = notifyChar || characteristic;
                    if (targetChar && typeof targetChar.stopNotifications === "function") {
                        await targetChar.stopNotifications();
                        console.log("✅ Stopped Bluetooth notifications");
                    }
                } catch (e) {
                    console.warn("⚠️ Stop notifications failed (safe to ignore):", e);
                }

                // Disconnect the device
                device.gatt.disconnect();
                console.log("🔴 Disconnected from device");
            } else {
                console.warn("⚠️ Device already disconnected");
            }
        } catch (err) {
            console.error("❌ Disconnect error:", err);
        } finally {
            handleDisconnectedState(window.bluetoothDevice);
        }
    };

    if (!open) return null;

    // ============================
    // UI
    // ============================
    return (
        <div style={styles.overlay}>
            <div style={styles.modal}>
                <div style={styles.header}>
                    <h3 style={styles.title}>
                        {!mode
                            ? "🔵 Connect Bluetooth Device"
                            : mode === "nano"
                            ? "AI & IOT Devices"
                            : "ESP32 Devices"}
                    </h3>
                    <button style={styles.closeBtn} onClick={onClose}>
                        ✖
                    </button>
                </div>

                <div style={styles.body}>
                    {!mode ? (
                        <>
                            <p>Select device type:</p>

                            <div style={styles.centerButtons}>
                                <button
                                    style={styles.actionBtn}
                                    onClick={() => handleModeSelect("nano")}
                                >
                                    AI & IOT (Nano)
                                </button>

                                <button
                                    style={styles.actionBtn}
                                    onClick={() => handleModeSelect("esp32")}
                                >
                                    AI & Robotics (ESP32)
                                </button>
                            </div>
                        </>
                    ) : (
                        <>
                            <button
                                style={styles.addBtn}
                                disabled={loading}
                                onClick={addOneDevice}
                            >
                                {loading ? "Scanning..." : "Scan Devices"}
                            </button>

                            {devices.length === 0 ? (
                                <p>No devices found</p>
                            ) : (
                                <ul style={styles.deviceList}>
                                    {devices.map((d) => (
                                        <li key={d.id} style={styles.deviceItem}>
                                            <strong style={styles.deviceName}>{d.name}</strong>

                                            {connectedId === d.id ? (
                                                <button
                                                    style={styles.disconnectBtn}
                                                    onClick={handleManualDisconnect}
                                                >
                                                    Disconnect
                                                </button>
                                            ) : (
                                                <button
                                                    style={styles.connectBtn}
                                                    disabled={connectingId === d.id}
                                                    onClick={() => connectDevice(d)}
                                                >
                                                    {connectingId === d.id
                                                        ? "Connecting..."
                                                        : "Connect"}
                                                </button>
                                            )}
                                        </li>
                                    ))}
                                </ul>
                            )}

                            <button style={styles.backBtn} onClick={() => setMode(null)}>
                                Back
                            </button>
                        </>
                    )}
                </div>
            </div>
        </div>
    );
};
