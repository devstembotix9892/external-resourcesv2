const MenuBarItemTooltip = ({
    children,
    className,
    enable,
    id,
    place = 'bottom'
}) => {
    if (enable) {
        return (
            <React.Fragment>
                {children}
            </React.Fragment>
        );
    }
    return (
        <ComingSoonTooltip
            className={classNames(styles.comingSoon, className)}
            place={place}
            tooltipClassName={styles.comingSoonTooltip}
            tooltipId={id}
        >
            {children}
        </ComingSoonTooltip>
    );
};


MenuBarItemTooltip.propTypes = {
    children: PropTypes.node,
    className: PropTypes.string,
    enable: PropTypes.bool,
    id: PropTypes.string,
    place: PropTypes.oneOf(['top', 'bottom', 'left', 'right'])
};

        if (typeof document === 'undefined') return;

    clearStickyBluetoothSession() {
        if (typeof window !== 'undefined') {
            window.bluetoothManualDisconnect = true;
            window.bluetoothStickySession = false;
            window.bluetoothAutoReconnectEnabled = false;
            window.bluetoothReconnectPromise = null;
            if (window.activeConnectionType === 'bluetooth') {
                window.activeConnectionType = null;
            }
        }
    }

    handleStickyBluetoothDisconnect() {
        this.clearStickyBluetoothSession();

        const activeDevice = this.bluetoothDevice ||
            (typeof window !== 'undefined' ? window.bluetoothDevice : null);

        if (activeDevice && activeDevice.gatt && activeDevice.gatt.connected) {
            activeDevice.gatt.disconnect();
        }

        this.bluetoothDevice = null;
        if (typeof window !== 'undefined') {
            window.bluetoothDevice = null;
            window.bluetoothCharacteristic = null;
            window.bluetoothNotifyChar = null;
        }

        this.setState({
            isBluetoothConnected: false,
            connectedDeviceName: '',
            deviceName: null,
            showBluetoothPopup: false
        });

        if (this.props.onDisconnect) {
            this.props.onDisconnect();
        }
        if (this.props.onBluetoothPortDisconnected) {
            this.props.onBluetoothPortDisconnected();
        }
    }

    handleConfirmSaveProject() {
        if (this.state.isSaveProjectInProgress) {
            return;
        }
        const projectName = (this.state.saveProjectName || '').trim() || 'Blockzie Project';
        const saveProjectCallback = this.state.pendingSaveProjectCallback;

        this.setState({ isSaveProjectInProgress: true });
        this.props.onSetProjectTitle(projectName);
        this.setState({
            isSaveProjectModalOpen: false,
            saveProjectName: projectName,
            pendingSaveProjectCallback: null
        }, () => {
            if (saveProjectCallback) {
                Promise.resolve(saveProjectCallback(projectName))
                    .catch(err => {
                        console.error('Error saving project:', err);
                    })
                    .finally(() => {
                        this.setState({ isSaveProjectInProgress: false });
                    });
            } else {
                this.setState({ isSaveProjectInProgress: false });
            }
        });
    }

    applyBlockColorTheme(themeName) {
        const palette = this.getBlockColorPalette(themeName);
        const categoryByTypePrefix = [
            ['motion_', 'motion'],
            ['looks_', 'looks'],
            ['sound_', 'sounds'],
            ['event_', 'event'],
            ['control_', 'control'],
            ['sensing_', 'sensing'],
            ['operator_', 'operators'],
            ['data_list', 'data_lists'],
            ['data_', 'data'],
            ['pen_', 'pen'],
            ['procedures_', 'more'],
            ['argument_', 'more'],
            ['arduino_', 'arduino'],
            ['threedphysics_', 'threedphysics'],
            ['threedglobalControl_', 'threedglobalControl'],
            ['threed_', 'threed']
        ];

        ScratchBlocks.Colours.overrideColours(palette);

        const applyPaletteToWorkspace = workspace => {
            if (!workspace || !workspace.getAllBlocks) return;
            workspace.getAllBlocks(false).forEach(block => {
                const match = categoryByTypePrefix.find(([prefix]) => block.type.indexOf(prefix) === 0);
                if (match && palette[match[1]] && block.setColour) {
                    const colors = palette[match[1]];
                    block.setColour(colors.primary, colors.secondary, colors.tertiary);
                    if (block.render) block.render();
                }
            });
        };

        const workspace = ScratchBlocks.getMainWorkspace && ScratchBlocks.getMainWorkspace();
        if (workspace) {
            applyPaletteToWorkspace(workspace);
            if (workspace.getFlyout && workspace.getFlyout() && workspace.getFlyout().getWorkspace) {
                applyPaletteToWorkspace(workspace.getFlyout().getWorkspace());
            }
            if (workspace.toolbox_ && workspace.toolbox_.refreshSelection) {
                workspace.toolbox_.refreshSelection();
            }
            if (workspace.refreshToolboxSelection_) {
                workspace.refreshToolboxSelection_();
            }
        }
    }

        this.bluetoothMonitorInterval = setInterval(() => {
            if (typeof window === 'undefined') return;

            const device = window.bluetoothDevice;
            const isGattConnected = !!(device && device.gatt && device.gatt.connected === true);
            const wasConnected = this.state.isBluetoothConnected;

            if (isGattConnected !== wasConnected) {
                // State update so menu-bar ma "Connect / Disconnect" proper dekhay.

                    isBluetoothConnected: isGattConnected,

                    connectedDeviceName: isGattConnected && device ?
                        (device.name || this.state.connectedDeviceName) : ''
                });

                // Jo icon stage par thi gayab (wasConnected true, now false)
                // to device ne proper cleanup + disconnect treat karo.
                if (wasConnected && !isGattConnected) {
                    try {
                        if (window.bluetoothCharacteristic && window.bluetoothCharacteristic.stopNotifications) {
                            window.bluetoothCharacteristic.stopNotifications().catch(() => { });
                        }
                    } catch (e) {
                        // ignore
                    }

                    window.bluetoothCharacteristic = null;
                    window.bluetoothResponseQueue = null;
                    window.bluetoothNotificationListener = null;

                    if (device && device.gatt && device.gatt.connected) {
                        try {
                            device.gatt.disconnect();
                        } catch (e) {
                            // ignore
                        }
                    }

                    window.bluetoothDevice = null;

                    if (this.props.onBluetoothPortDisconnected) {
                        this.props.onBluetoothPortDisconnected();
                    }
                }
            }

    async silentDisconnectWiFi () {
        try {
            const ip = window.esp32IP || this.state.esp32IP || '192.168.4.1';
            const isLocalDevHost = window.location.hostname === 'localhost' ||
                window.location.hostname === '127.0.0.1';
            const useProxy = isLocalDevHost && !isScratchDesktop();
            const useHttpControl = isScratchDesktop() || window.esp32UseHttpControl === true;
            const protocol = useHttpControl ? 'http' : 'https';
            const disconnectURL = useProxy ?
                `/esp32-proxy/${ip}/disconnect` :
                `${protocol}://${ip}/disconnect`;

            const response = await fetch(disconnectURL, {method: 'POST'});
            if (!response.ok && response.status !== 404) {
                throw new Error('Failed to disconnect');
            }
        } catch (error) {
            // Ignore WiFi disconnect errors during project reset.
        } finally {
            this.setState({isWiFiConnected: false, esp32IP: null});
            if (typeof window !== 'undefined') {
                window.isWiFiConnected = false;
                window.esp32IP = null;
                if (window.__esp32FetchPatched && window.__esp32FetchOriginal) {
                    window.fetch = window.__esp32FetchOriginal;
                    window.__esp32FetchPatched = false;
                }
            }
            if (typeof localStorage !== 'undefined') {
                localStorage.removeItem('esp32IP');
            }
        }
    }
    silentDisconnectBluetooth () {
        this.clearStickyBluetoothSession();

        const activeDevice = this.bluetoothDevice ||
            (typeof window !== 'undefined' ? window.bluetoothDevice : null);

        if (activeDevice && activeDevice.gatt && activeDevice.gatt.connected) {
            try {
                if (window.bluetoothCharacteristic && window.bluetoothCharacteristic.stopNotifications) {
                    window.bluetoothCharacteristic.stopNotifications().catch(() => {});
                }
            } catch (e) {
                // ignore
            }

            try {
                activeDevice.gatt.disconnect();
            } catch (e) {
                // ignore
            }
        }

        this.bluetoothDevice = null;
        if (typeof window !== 'undefined') {
            window.bluetoothDevice = null;
            window.bluetoothCharacteristic = null;
            window.bluetoothNotifyChar = null;
            window.bluetoothResponseQueue = null;
            window.bluetoothNotificationListener = null;
            if (window.activeConnectionType === 'bluetooth') {
                window.activeConnectionType = null;
            }
        }

        this.setState({
            isBluetoothConnected: false,
            connectedDeviceName: '',
            deviceName: null,
            showBluetoothPopup: false
        });
    }
    async resetSessionForNewProject () {
        const {vm, deviceId, peripheralName} = this.props;

        vm.stopAll();

        if (peripheralName && deviceId) {
            try {
                vm.disconnectPeripheral(deviceId);
            } catch (e) {
                // ignore
            }
        }

        if (this.state.isBluetoothConnected ||
            (typeof window !== 'undefined' && window.bluetoothDevice)) {
            this.silentDisconnectBluetooth();
        }

        if (this.state.isWiFiConnected ||
            (typeof window !== 'undefined' && window.isWiFiConnected)) {
            await this.silentDisconnectWiFi();
        }

        try {
            await vm.extensionManager.loadDeviceURL('unselectDevice', 'arduino');
        } catch (e) {
            vm.extensionManager.clearDevice();
        }

        if (typeof window !== 'undefined') {
            window.activeConnectionType = null;
        }

        if (vm && vm.runtime) {
            vm.runtime.isThreeDProject = false;
            vm.runtime.isPythonProject = false;
            vm.runtime.isBlocksProject = true;
            vm.runtime.__skipAutoThreeTabOnce = false;
        }

        this.props.onClearProjectSessionState();
    }

   connectToESP32 = async (retryCount = 0, maxRetries = 2) => {
        try {
            const device = await navigator.bluetooth.requestDevice({
                acceptAllDevices: true,
                optionalServices: ["d804b643-6ce7-4e81-9f8a-ce0f699085eb"],
            });

            console.log("📱 Device selected:", device.name);

            // Wait for device to be ready
            await new Promise(resolve => setTimeout(resolve, 500));

            const server = await device.gatt.connect();
            console.log("🔗 GATT connected");

            // Wait for GATT to settle
            await new Promise(resolve => setTimeout(resolve, 300));

            this.bluetoothDevice = device;

            // Store globally for stage header detection
            window.bluetoothDevice = device;

            // ✅ Store the device ID for firmware upload
            this.setState({
                isBluetoothConnected: true,
                deviceName: device.name || "Unknown Device",
            });

            // Show connection alert
            if (this.props.onBluetoothPortConnected) {
                this.props.onBluetoothPortConnected();
            }

            // ✅ Update deviceId in props (if available)
            if (this.props.onPeripheralConnected) {
                this.props.onPeripheralConnected(device.name || "ESP32", device.id);
            }

            const service = await server.getPrimaryService("d804b643-6ce7-4e81-9f8a-ce0f699085eb");
            const characteristic = await service.getCharacteristic("c8659211-af91-4ad3-a995-a58d6fd26145");

            window.bluetoothCharacteristic = characteristic;

            // Initialize response queue
            window.bluetoothResponseQueue = [];

            // Set up notification listener for receiving responses
            try {
                await characteristic.startNotifications();
                console.log("✅ Bluetooth notifications started");

                characteristic.addEventListener('characteristicvaluechanged', (event) => {
                    const value = new TextDecoder().decode(event.target.value);
                    console.log("📥 Bluetooth Data Received:", value);

                    // Store response in queue
                    if (!window.bluetoothResponseQueue) {
                        window.bluetoothResponseQueue = [];
                    }
                    window.bluetoothResponseQueue.push(value);
                });

                console.log("✅ Bluetooth notification listener set up");
            } catch (notifError) {
                console.warn("⚠ Could not start Bluetooth notifications:", notifError);
                // Continue anyway - some devices might not support notifications
            }
        } catch (error) {
            console.error("Bluetooth connect failed:", error);

            // Retry logic for connection errors (but not for user cancellation)
            if (retryCount < maxRetries &&
                error.name !== "NotFoundError" &&
                !error.message.includes("cancelled")) {
                console.log(`🔄 Retrying connection (attempt ${retryCount + 2}/${maxRetries + 1})...`);
                // Wait before retrying
                await new Promise(resolve => setTimeout(resolve, 1000));
                return this.connectToESP32(retryCount + 1, maxRetries);
            }

            this.setState({ isBluetoothConnected: false });
            if (error.name === "NotFoundError" || error.message.includes("cancelled")) {
                this.props.onBluetoothPortDisconnected(); // 🔥 Trigger when user cancels
            }
        }
    };


    disconnectFromESP32 = async () => {

        if (this.bluetoothDevice && this.bluetoothDevice.gatt.connected) {
            // Clean up notification listener
            if (window.bluetoothCharacteristic) {
                try {
                    await window.bluetoothCharacteristic.stopNotifications();
                    console.log("✅ Bluetooth notifications stopped");
                } catch (err) {
                    console.warn("⚠ Error stopping notifications:", err);
                }
            }

            this.bluetoothDevice.gatt.disconnect();

            // 🔹 Reset the state and global variables
            window.bluetoothCharacteristic = null;
            window.bluetoothResponseQueue = null;
            window.bluetoothNotificationListener = null;
            window.bluetoothDevice = null;
            this.setState({ isBluetoothConnected: false });

            // Show disconnection alert
            if (this.props.onBluetoothPortDisconnected) {
                this.props.onBluetoothPortDisconnected();
            }
        } else {
            console.warn("⚠ No active Bluetooth connection to disconnect.");
        }
    };

    connectToNanoHM10 = async () => {
    try {
        const device = await navigator.bluetooth.requestDevice({
            acceptAllDevices: true,
            optionalServices: ["0000ffe0-0000-1000-8000-00805f9b34fb"]
        });

        const server = await device.gatt.connect();
        const service = await server.getPrimaryService("0000ffe0-0000-1000-8000-00805f9b34fb");

        // ✅ Write માટે ffe2
        const writeChar = await service.getCharacteristic("0000ffe2-0000-1000-8000-00805f9b34fb");
        
        // ✅ Read/Notify માટે ffe1

        // ✅ Global save
        window.bluetoothCharacteristic = writeChar;      // write માટે
        window.bluetoothNotifyChar = notifyChar;         // response માટે

        // ✅ Notifications start કરો
        await notifyChar.startNotifications();
        notifyChar.addEventListener("characteristicvaluechanged", (event) => {
            const value = new TextDecoder().decode(event.target.value);
            console.log("📥 BLE Response:", value);
            window.bluetoothLastResponse = (window.bluetoothLastResponse || "") + value;
        });

        window.bluetoothDevice = device;
        this.bluetoothDevice = device;
        this.bluetoothCharacteristic = writeChar;

        this.setState({
            isBluetoothConnected: true,
            deviceName: device.name
        });

        if (this.props.onBluetoothPortConnected) {
            this.props.onBluetoothPortConnected();
        }

        console.log("✅ Connected to device:", device.name);
        console.log("✅ Characteristic ready:", writeChar.uuid);
        return writeChar;

    } catch (err) {
        console.error("Bluetooth connect failed:", err);
        this.setState({
            isBluetoothConnected: false,
            deviceName: "Disconnected"
        });
        if (err.name === "NotFoundError" || err.message.includes("cancelled")) {
            this.props.onBluetoothPortDisconnected();
        }
        return null;
    }

    handleWiFiConnect = async (nextIp) => {
        try {
            let ip = (nextIp || '').trim() || localStorage.getItem("ipAddress");

            if (!ip) {
                ip = prompt("📡 Enter the IP address of the ESP32:", "192.168.4.1");
                if (!ip) throw new Error("No IP entered");
                localStorage.setItem("ipAddress", ip);
            }

            console.log("Connecting to IP:", ip);

            // Browser on localhost needs WDS /esp32-proxy (CORS + mixed content). Electron desktop talks to ESP32 directly.
            const isLocalDevHost = window.location.hostname === 'localhost' ||
                window.location.hostname === '127.0.0.1';
            const useProxy = isLocalDevHost && !isScratchDesktop();
            // Blockzie HTTP firmware uses port 80; HTTPS firmware uses 443 (window.esp32UseHttpControl === false).
            const useHttpControl = isScratchDesktop() || window.esp32UseHttpControl === true;
            const protocol = useHttpControl ? 'http' : 'https';
            const connectURL = useProxy ? `/esp32-proxy/${ip}/connect` : `${protocol}://${ip}/connect`;

            const controller = new AbortController();
            const timeoutId = setTimeout(() => controller.abort(), 8000);

            const response = await fetch(connectURL, { signal: controller.signal });
            clearTimeout(timeoutId);

            // 200 = /connect exists. 404 = Blockzie firmware only has /control, no /connect - still OK, device is reachable
            if (!response.ok && response.status !== 404) throw new Error("Connection failed");

            const text = await response.text();
            if (response.status === 404) {
                console.log("✅ Connected via WiFi! (Blockzie firmware uses /control only)");
            } else {
                console.log("✅ Connected via WiFi! Message:", text);
            }

            this.setState({
                isWiFiConnected: true,
                deviceName: ` WiFi @ ${ip}`,
                esp32IP: ip
            });

            // Save globally so VM uses WiFi for realtime commands (not Bluetooth)
            window.activeConnectionType = 'wifi';
            window.isWiFiConnected = true;
            window.esp32IP = ip;
            window.esp32UseHttpControl = useHttpControl;
            if (typeof window.esp32WifiUseHttp !== 'undefined') {
                delete window.esp32WifiUseHttp;
            }
            // Browser localhost only: rewrite http(s)://<ip>/control → /esp32-proxy/... (see scripts/esp32-dev-proxy.js).
            if (useProxy) {
                if (!window.__esp32FetchOriginal) {
                    window.__esp32FetchOriginal = window.fetch.bind(window);
                }
                if (!window.__esp32FetchPatched) {
                    window.fetch = function (input, init) {
                        try {
                            const rawUrl = typeof input === 'string' ? input : (input && input.url);
                            if (typeof rawUrl === 'string') {
                                const rewrittenUrl = rawUrl.replace(
                                    /^https?:\/\/(\d{1,3}(?:\.\d{1,3}){3})(?::\d+)?\/control(.*)$/i,
                                    '/esp32-proxy/$1/control$2'
                                );
                                if (rewrittenUrl !== rawUrl) {
                                    return window.__esp32FetchOriginal(rewrittenUrl, init);
                                }
                            }
                        } catch (e) {
                            // Fall through to original fetch if rewrite fails.
                        }
                        return window.__esp32FetchOriginal(input, init);
                    };
                    window.__esp32FetchPatched = true;
                }
            }
            window.bluetoothCharacteristic = null; // Prefer WiFi; clear stale BLE so VM doesn't try BLE first

            if (this.props.onWiFiConnected) {
                this.props.onWiFiConnected();
            }
            if (this.props.onPeripheralConnected) {
                this.props.onPeripheralConnected("ESP32 (WiFi)", "WIFI-" + ip);
            }

        } catch (error) {
            console.error("❌ Failed to connect via WiFi:", error);
            this.setState({ isWiFiConnected: false });
            window.isWiFiConnected = false;
        }
    };

    handleWiFiDisconnect = async () => {
        try {
            const ip = window.esp32IP || this.state.esp32IP || "192.168.4.1";

            const isLocalDevHost = window.location.hostname === 'localhost' ||
                window.location.hostname === '127.0.0.1';
            const useProxy = isLocalDevHost && !isScratchDesktop();
            const useHttpControl = isScratchDesktop() || window.esp32UseHttpControl === true;
            const protocol = useHttpControl ? 'http' : 'https';
            const disconnectURL = useProxy ? `/esp32-proxy/${ip}/disconnect` : `${protocol}://${ip}/disconnect`;

            const response = await fetch(disconnectURL, {
                method: 'POST'
            });

            // Some firmwares (like https-only) may not implement /disconnect.
            // 404 still means the device is reachable; don't fail hard.
            if (!response.ok && response.status !== 404) throw new Error("Failed to disconnect");
            console.log("✅ ESP32 disconnected WiFi session.");
        } catch (error) {
            console.error("❌ WiFi Disconnect failed:", error);
        } finally {
            this.setState({ isWiFiConnected: false, esp32IP: null });
            window.isWiFiConnected = false;
            window.esp32IP = null;
            localStorage.removeItem("esp32IP");
            if (window.__esp32FetchPatched && window.__esp32FetchOriginal) {
                window.fetch = window.__esp32FetchOriginal;
                window.__esp32FetchPatched = false;
            }

            // Show disconnection alert
            if (this.props.onWiFiDisconnected) {
                this.props.onWiFiDisconnected();
            }
        }
    };

    async handleProgramModeUpdate(data) {
        const isLockedWorkspace =
            Boolean(this.props.deviceId) &&
            (this.props.activeTabIndex === 3 ||
                this.props.activeTabIndex === 4);

        if (!data.isRealtimeMode && isLockedWorkspace) {
            return;
        }

        if (data.isRealtimeMode) {
            this.props.onSetRealtimeMode();
        } else {
            this.props.onSetUploadMode();

            // Ensure any connect sub-UI is closed in upload mode.
            if (this.state.showBluetoothPopup) {
                this.setState({ showBluetoothPopup: false });
            }

            // USB serial auto-disconnects at OS level when the board
            // resets or when we leave realtime mode. To mirror that
            // behaviour for Bluetooth, explicitly drop any active
            // Bluetooth connections whenever we switch OUT of
            // realtime mode (i.e. into upload mode).
            if (this.state.isBluetoothConnected) {
                try {
                    await this.disconnectFromESP32();
                } catch (e) {
                    // Ignore – ESP32 BLE may not be the active profile.
                    // eslint-disable-next-line no-console
                    console.warn('Bluetooth ESP32 auto-disconnect failed:', e);
                }
                try {
                    await this.disconnectFromNanoHM10();
                } catch (e) {
                    // Ignore – Nano/HM10 may not be connected.
                    // eslint-disable-next-line no-console
                    console.warn('Bluetooth Nano/HM10 auto-disconnect failed:', e);
                }
            }
            // WiFi is only supported for realtime control. Drop any active WiFi session when entering upload mode.
            if (this.state.isWiFiConnected) {
                try {
                    await this.handleWiFiDisconnect();
                } catch (e) {
                    // eslint-disable-next-line no-console
                    console.warn('WiFi auto-disconnect failed:', e);
                }
            }
        }
    }

    handleConfirmOpenProjectFromLink = projectId => {
        const rawValue = (projectId || '').trim();
        if (!rawValue) return;

        const normalizeProjectLink = value => {
            try {
                if (/^[a-z]+:\/\//i.test(value)) {
                    return value;
                }

                if (/^(?:www\.)?(?:drive\.google\.com|docs\.google\.com|googleusercontent\.com)\//i.test(value)) {
                    return `https://${value}`;
                }

                return value;
            } catch (e) {
                return value;
            }
        };

        const getDriveDownloadUrl = value => {
            try {
                const normalizedValue = normalizeProjectLink(value);
                const url = new URL(normalizedValue);
                const host = url.hostname.toLowerCase();
                if (!host.includes('drive.google.com') && !host.includes('docs.google.com')) {
                    return normalizedValue;
                }

                const fileIdMatch = normalizedValue.match(/\/file\/d\/([^/?#]+)/i) ||
                    normalizedValue.match(/[?&]id=([^&#]+)/i);
                if (!fileIdMatch) return normalizedValue;
                return `https://drive.google.com/uc?export=download&id=${encodeURIComponent(fileIdMatch[1])}`;
            } catch (e) {
                return normalizeProjectLink(value);
            }
        };

        const isProjectFileLink = (() => {
            try {
                const url = new URL(normalizeProjectLink(rawValue));
                const host = url.hostname.toLowerCase();
                return /\.(?:ob|sb|sb2|sb3)$/i.test(url.pathname) ||
                    host.includes('drive.google.com') ||
                    host.includes('docs.google.com');
            } catch (e) {
                return /\.(?:ob|sb|sb2|sb3)$/i.test(rawValue);
            }
        })();

        if (isProjectFileLink) {
            const downloadUrl = getDriveDownloadUrl(rawValue);
            const isLocalDevHost = window.location.hostname === 'localhost' ||
                window.location.hostname === '127.0.0.1';
            const useWebpackProxy = isLocalDevHost && !isScratchDesktop();
            const useDesktopFetch = window.blockzieDesktopFetch &&
                typeof window.blockzieDesktopFetch.fetchProjectFileUrl === 'function';

            const fetchProjectData = () => {
                if (useDesktopFetch) {
                    return window.blockzieDesktopFetch.fetchProjectFileUrl(
                        normalizeProjectLink(rawValue)
                    );
                }

                const requestUrl = useWebpackProxy
                    ? `/project-file-proxy?url=${encodeURIComponent(downloadUrl)}`
                    : downloadUrl;

                return fetch(requestUrl).then(async response => {
                    if (!response.ok) {
                        const responseText = await response.text().catch(() => '');
                        const trimmedText = responseText ? responseText.slice(0, 180) : '';
                        throw new Error(
                            `Failed to fetch project file: ${response.status}` +
                            (trimmedText ? ` - ${trimmedText}` : '')
                        );
                    }
                    return response.arrayBuffer();
                });
            };

            fetchProjectData()
                .then(projectData => {
                    if (!this.props.vm || typeof this.props.vm.loadProject !== 'function') {
                        throw new Error('Project loader is unavailable.');
                    }
                    return this.props.vm.loadProject(projectData).then(() =>
                        this.applyProjectLinkTabState(projectData)
                    );
                })
                .then(() => {
                    this.closeProjectLinkModal();
                })
                .catch(error => {
                    // eslint-disable-next-line no-console
                    console.error('Unable to open project from link:', error);
                    if (typeof window !== 'undefined' && typeof window.alert === 'function') {
                        window.alert(error && error.message
                            ? `Unable to open the project from this link.\n${error.message}`
                            : 'Unable to open the project from this link.');
                    }
                });
            return;
        }

        const projectIdMatch = rawValue.match(/(?:projectId=|\/projects\/)([^/?#]+)/i);
        const projectIdValue = projectIdMatch ? projectIdMatch[1] : rawValue;
        this.closeProjectLinkModal();
        this.navigateToPath(`/editor?projectId=${encodeURIComponent(projectIdValue)}`);
    };

const mapStateToProps = (state, ownProps) => {
    const loadingState = state.scratchGui.projectState.loadingState;
    const user = state.session && state.session.session && state.session.session.user;
    return {
        activeTabIndex: state.scratchGui.editorTab.activeTabIndex,
        aboutMenuOpen: aboutMenuOpen(state),
        accountMenuOpen: accountMenuOpen(state),
        fileMenuOpen: fileMenuOpen(state),
        settingMenuOpen: settingMenuOpen(state),
        editMenuOpen: editMenuOpen(state),
        connectMenuOpen: connectMenuOpen(state),
        isUpdating: getIsUpdating(loadingState),
        isRealtimeMode: state.scratchGui.programMode.isRealtimeMode,
        isRtl: state.locales.isRtl,
        isShowingProject: getIsShowingProject(loadingState),
        isSupportSwitchMode: state.scratchGui.programMode.isSupportSwitchMode,
        isToolboxUpdating: state.scratchGui.toolbox.isToolboxUpdating,
        languageMenuOpen: languageMenuOpen(state),
        locale: state.locales.locale,
        messagesByLocale: state.locales.messagesByLocale,
        loginMenuOpen: loginMenuOpen(state),
        projectTitle: state.scratchGui.projectTitle,
        realtimeConnection: state.scratchGui.connectionModal.realtimeConnection,
        sessionExists: state.session && typeof state.session.session !== 'undefined',
        username: user ? user.username : null,
        userOwnsProject: ownProps.authorUsername && user &&
            (ownProps.authorUsername === user.username),
        stageSizeMode: state.scratchGui.stageSize.stageSize,
        vm: state.scratchGui.vm,
        peripheralName: state.scratchGui.connectionModal.peripheralName,
        showConnectionMethodModal: state.scratchGui.connectionModal.showConnectionMethodModal,
        deviceId: state.scratchGui.device.deviceId,
        deviceName: state.scratchGui.device.deviceName,
        allFiles: state.scratchGui.filesData.allFiles,

        macMenuOpen: state.scratchGui.menus.macMenuOpen || false,

    };
};
