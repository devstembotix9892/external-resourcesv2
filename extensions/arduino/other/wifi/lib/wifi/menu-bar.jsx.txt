    children,
    className,
    enable,
    id,
    place = 'bottom'
}) => {
    if (enable) {
        return (
            <React.Fragment>
                {children}
            </React.Fragment>
        );
    }
    return (
        <ComingSoonTooltip
            className={classNames(styles.comingSoon, className)}
            place={place}
            tooltipClassName={styles.comingSoonTooltip}
            tooltipId={id}

            {children}
        </ComingSoonTooltip>
    );
};

        return Boolean(

        if (typeof window === 'undefined' || !window.localStorage) {
            return '';

    renderBlockzieAuthControls () {
        if (!isScratchDesktop()) {
            return null;
        }

        const loggedIn = this.state.blockzieLoggedIn;
        const username = this.getBlockzieUsername();

        return (
            <>
                {loggedIn && username ? (
                    <div className={classNames(styles.menuBarItem, styles.blockzieUserName)}>
                        {username}
                    </div>
                ) : null}
                <div className={classNames(styles.menuBarItem, styles.hoverable)}>
                    <button
                        type="button"
                        className={loginButtonStyles.loginButtonMenubar}
                        data-blockzie-auth-button="true"
                        onClick={() => {
                            if (loggedIn) {
                                if (typeof window.blockzieLogout === 'function') {
                                    window.blockzieLogout();
                                }
                                return;
                            }
                            console.log('[Blockzie] Login button clicked');
                            if (typeof window.blockzieOpenKeycloakLogin === 'function') {
                                window.blockzieOpenKeycloakLogin();
                            } else if (this.props.onClickLogin) {
                                this.props.onClickLogin();
                            } else {
                                try {
                                    const {ipcRenderer} = window.require('electron');
                                    ipcRenderer.send('open-keycloak-login-window');
                                } catch (ipcErr) {
                                    console.error('[Blockzie] Login handler missing', ipcErr);
                                }
                            }
                        }}
                    >
                        {loggedIn ? 'Logout' : 'Login'}
                    </button>
                </div>

        const next = this.getBlockzieLoggedIn();

            this.forceUpdate();

    connectToNanoHM10 = async () => {
            try {
                // Step 1: Device select
                const device = await navigator.bluetooth.requestDevice({
                    acceptAllDevices: true,
                    optionalServices: ["0000ffe0-0000-1000-8000-00805f9b34fb"]
                });
    
                // Step 2: Connect GATT
                const server = await device.gatt.connect();
    
                // Step 3: Get service & characteristic
                const service = await server.getPrimaryService("0000ffe0-0000-1000-8000-00805f9b34fb");
                const characteristic = await service.getCharacteristic("0000ffe2-0000-1000-8000-00805f9b34fb");
    
                // Step 4: Save globally for block access
                window.bluetoothDevice = device;
                window.bluetoothCharacteristic = characteristic;
    
                this.bluetoothCharacteristic = characteristic;
    
                // Step 5: Auto disconnect handler
                // device.addEventListener("gattserverdisconnected", () => {
                //     console.warn("❌ Bluetooth disconnected");
                //     this.bluetoothCharacteristic = null;
                //     this.setState({
                //         isBluetoothConnected: false,
                //         deviceName: "Disconnected"
                //     });
                // });
    
                // Step 6: Update component state
                // Step 6: Update component state
                this.bluetoothDevice = device;
                this.setState({
                    isBluetoothConnected: true,
                    deviceName: device.name
                });
    
                console.log("✅ Connected to device:", device.name);
                console.log("✅ Characteristic ready:", characteristic.uuid);
                this.setState({
                    isBluetoothConnected: true,
                    deviceName: device.name
                });
                return characteristic;
    
            } catch (err) {
                console.error("Bluetooth connect failed:", err);
                this.setState({
                    isBluetoothConnected: false,
                    deviceName: "Disconnected"
                });
                if (err.name === "NotFoundError" || err.message.includes("cancelled")) {
                    this.props.onBluetoothPortDisconnected();
                }
                return null;
            }
        };
    



    // Disconnect Nano HM-10
    disconnectFromNanoHM10 = async () => {
        try {
            const device = this.bluetoothDevice || this.state.bluetoothDevice;
            if (device) {
                if (device.gatt.connected) {
                    await device.gatt.disconnect();
                    console.log("✅ Disconnected from Arduino Nano (HM-10).");
                } else {
                    console.log("ℹ Already disconnected.");
                }

                this.setState({
                    isBluetoothConnected: false,
                    deviceName: "Disconnected",
                    bluetoothDevice: null
                });

                this.bluetoothDevice = null;
                this.bluetoothCharacteristic = null;
            } else {
                console.warn("⚠ No Nano device found to disconnect.");
            }
        } catch (error) {
            console.error("❌ Error disconnecting Nano:", error);
        }
    };

    // Send data to Nano via Bluetooth
    sendNanoCommand = async (command) => {
        if (window.bluetoothCharacteristic) {
            try {
                const encoder = new TextEncoder();
                await window.bluetoothCharacteristic.writeValue(encoder.encode(command));
                console.log("✅ Sent to Nano:", command);
            } catch (err) {
                console.error("❌ Bluetooth Error:", err);
            }
        } else {
            console.warn("⚠️ No Nano Bluetooth connection available.");
        }
    };

            this.props.onRequestCloseConnect();
        } else {
            await this.disconnectFromNanoHM10();
            this.props.onRequestCloseConnect();
        }
    };

    readBluetoothGapName = async device => {
        if (!device?.gatt) return '';

        const wasConnected = Boolean(device.gatt.connected);

        try {
            const server = wasConnected ? device.gatt : await device.gatt.connect();
            const service = await server.getPrimaryService('generic_access');

            try {
                const characteristic = await service.getCharacteristic('gap.device_name');
                const value = await characteristic.readValue();
                return new TextDecoder('utf-8').decode(value).trim();
            } catch (error) {
                const characteristic = await service.getCharacteristic(0x2a00);
                const value = await characteristic.readValue();
                return new TextDecoder('utf-8').decode(value).trim();
            }
        } catch (error) {
            console.warn('Could not read GAP device name:', error);
            return '';
        } finally {
            if (!wasConnected && device.gatt?.connected) {
                device.gatt.disconnect();
            }
        }
    };

    resolveBluetoothDeviceName = async (device, fallbackName) => {
        const directName = (device?.name || '').trim();
        if (directName && !/^(unknown|unnamed|device|bluetooth device|n\/a)$/i.test(directName)) {
            return directName;
        }

        const gapName = await this.readBluetoothGapName(device);
        if (gapName && !/^(unknown|unnamed|device|bluetooth device|n\/a)$/i.test(gapName)) {
            return gapName;
        }

        return fallbackName;
    };

    connectToESP32 = async (retryCount = 0, maxRetries = 2) => {
        try {
            const device = await navigator.bluetooth.requestDevice({
                acceptAllDevices: true,
                optionalServices: ["d804b643-6ce7-4e81-9f8a-ce0f699085eb"],
            });

            console.log("📱 Device selected:", device.name);

            // Wait for device to be ready
            await new Promise(resolve => setTimeout(resolve, 500));

            const server = await device.gatt.connect();
            console.log("🔗 GATT connected");

            // Wait for GATT to settle
            await new Promise(resolve => setTimeout(resolve, 300));

            const resolvedDeviceName = await this.resolveBluetoothDeviceName(device, "AI & Robotics");

            this.bluetoothDevice = device;

            // Store globally for stage header detection
            window.bluetoothDevice = device;

            // ✅ Store the device ID for firmware upload
            this.setState({
                isBluetoothConnected: true,
                deviceName: resolvedDeviceName,
            });

            // Show connection alert
            if (this.props.onBluetoothPortConnected) {
                this.props.onBluetoothPortConnected();
            }

            this.keepAliveInterval = setInterval(async () => {
                if (!this.bluetoothDevice?.gatt?.connected || !this.bluetoothCharacteristic) {
                    this.clearNanoKeepAlive();
                    return;
                }

                try {
                    await this.bluetoothCharacteristic.writeValue(
                        new TextEncoder().encode("ping\n")
                    );
                } catch (keepAliveError) {
                    console.warn("HM-10 keep alive failed:", keepAliveError);
                }
            }, 3000);

            this.keepAliveInterval = setInterval(async () => {
                if (!this.bluetoothDevice?.gatt?.connected || !this.bluetoothCharacteristic) {
                    this.clearNanoKeepAlive();
                    return;
                }

                try {
                    await this.bluetoothCharacteristic.writeValue(
                        new TextEncoder().encode("ping\n")
                    );
                } catch (keepAliveError) {
                    console.warn("HM-10 keep alive failed:", keepAliveError);
                }
            }, 3000);

            // ✅ Update deviceId in props (if available)
            if (this.props.onPeripheralConnected) {
                this.props.onPeripheralConnected(resolvedDeviceName, device.id);
            }

            const service = await server.getPrimaryService("d804b643-6ce7-4e81-9f8a-ce0f699085eb");
            const characteristic = await service.getCharacteristic("c8659211-af91-4ad3-a995-a58d6fd26145");

            window.bluetoothCharacteristic = characteristic;

            // Initialize response queue
            window.bluetoothResponseQueue = [];

            // Set up notification listener for receiving responses
            try {
                await characteristic.startNotifications();
                console.log("✅ Bluetooth notifications started");

                characteristic.addEventListener('characteristicvaluechanged', (event) => {
                    const value = new TextDecoder().decode(event.target.value);
                    console.log("📥 Bluetooth Data Received:", value);

                    // Store response in queue
                    if (!window.bluetoothResponseQueue) {
                        window.bluetoothResponseQueue = [];
                    }
                    window.bluetoothResponseQueue.push(value);
                });

                console.log("✅ Bluetooth notification listener set up");
            } catch (notifError) {
                console.warn("⚠ Could not start Bluetooth notifications:", notifError);
                // Continue anyway - some devices might not support notifications
            }
        } catch (error) {
            console.error("Bluetooth connect failed:", error);

            // Retry logic for connection errors (but not for user cancellation)
            if (retryCount < maxRetries &&
                error.name !== "NotFoundError" &&
                !error.message.includes("cancelled")) {
                console.log(`🔄 Retrying connection (attempt ${retryCount + 2}/${maxRetries + 1})...`);
                // Wait before retrying
                await new Promise(resolve => setTimeout(resolve, 1000));
                return this.connectToESP32(retryCount + 1, maxRetries);
            }

            this.setState({ isBluetoothConnected: false });
            if (error.name === "NotFoundError" || error.message.includes("cancelled")) {
                this.props.onBluetoothPortDisconnected(); // 🔥 Trigger when user cancels
            }
        }
    };

    handleWiFiConnect = async (nextIp) => {
        try {
            let ip = nextIp || localStorage.getItem('ipAddress');

            if (!ip) {
                ip = prompt("📡 Enter the IP address of the ESP32:", "192.168.4.1");
                if (!ip) throw new Error('No IP entered');
            }

            localStorage.setItem('ipAddress', ip);

            console.log("Connecting to IP:", ip);

            // Use proxy on localhost to avoid CORS; direct URL otherwise
            const useProxy = window.location.hostname === 'localhost' || window.location.hostname === '127.0.0.1';
            const connectURL = useProxy ? `/esp32-proxy/${ip}/connect` : `http://${ip}/connect`;

            const controller = new AbortController();
            const timeoutId = setTimeout(() => controller.abort(), 8000);

            const response = await fetch(connectURL, { signal: controller.signal });
            clearTimeout(timeoutId);

            // 200 = /connect exists. 404 = Blockzie firmware only has /control, no /connect - still OK, device is reachable
            if (!response.ok && response.status !== 404) throw new Error("Connection failed");

            const text = await response.text();
            if (response.status === 404) {
                console.log("✅ Connected via WiFi! (Blockzie firmware uses /control only)");
            } else {
                console.log("✅ Connected via WiFi! Message:", text);
            }

            this.setState({
                isWiFiConnected: true,
                deviceName: ` WiFi @ ${ip}`,
                esp32IP: ip
            });

            // Save globally so VM uses WiFi for realtime commands (not Bluetooth)
            window.activeConnectionType = 'wifi';
            window.isWiFiConnected = true;
            window.esp32IP = ip;
            window.bluetoothCharacteristic = null; // Prefer WiFi; clear stale BLE so VM doesn't try BLE first

            if (this.props.onWiFiConnected) {
                this.props.onWiFiConnected();
            }
            if (this.props.onPeripheralConnected) {
                this.props.onPeripheralConnected('ESP32 (WiFi)', `WIFI-${ip}`);
            }

        } catch (error) {
            console.error("❌ Failed to connect via WiFi:", error);
            this.setState({ isWiFiConnected: false });
            window.isWiFiConnected = false;
        }
    };

    handleWiFiDisconnect = async () => {
        try {
            const ip = window.esp32IP || this.state.esp32IP || '192.168.4.1';

            const useProxy = window.location.hostname === 'localhost' || window.location.hostname === '127.0.0.1';
            const disconnectURL = useProxy ? `/esp32-proxy/${ip}/disconnect` : `http://${ip}/disconnect`;

            const response = await fetch(disconnectURL, {
                method: 'POST'
            });

            if (!response.ok) throw new Error("Failed to disconnect");
            console.log("✅ ESP32 disconnected WiFi session.");
        } catch (error) {
            console.error("❌ WiFi Disconnect failed:", error);
        } finally {
            this.setState({ isWiFiConnected: false, esp32IP: null });
            if (window.activeConnectionType === 'wifi') {
                window.activeConnectionType = null;
            }
            window.isWiFiConnected = false;
            window.esp32IP = null;
            localStorage.removeItem('ipAddress');

            // Show disconnection alert
            if (this.props.onWiFiDisconnected) {
                this.props.onWiFiDisconnected();
            }
        }
    };

    disconnectFromESP32 = async () => {
        try {
            // Use whichever reference exists
            const device =
                this.bluetoothDevice ||
                (typeof window !== "undefined" ? window.bluetoothDevice : null);

            const characteristic =
                this.bluetoothCharacteristic ||
                (typeof window !== "undefined" ? window.bluetoothCharacteristic : null);

                        await characteristic.stopNotifications();

                    console.warn("stopNotifications failed (safe to ignore):", e);

                // disconnect
                device.gatt.disconnect();

        } catch (e) {
            console.error("Disconnect error:", e);
        } finally {
            // cleanup local + global refs
            this.bluetoothDevice = null;
            this.bluetoothCharacteristic = null;
            if (typeof window !== "undefined") {
                window.bluetoothDevice = null;
                window.bluetoothCharacteristic = null;

            // update UI + notify parent
            this.setState({
                isBluetoothConnected: false,
                connectedDeviceName: ""
            });
            if (this.props.onDisconnected) this.props.onDisconnected();
        }

const mapStateToProps = (state, ownProps) => {
    const loadingState = state.scratchGui.projectState.loadingState;
    const user = state.session && state.session.session && state.session.session.user;
    return {
        activeTabIndex: state.scratchGui.editorTab.activeTabIndex,
        aboutMenuOpen: aboutMenuOpen(state),
        accountMenuOpen: accountMenuOpen(state),
        fileMenuOpen: fileMenuOpen(state),
        settingMenuOpen: settingMenuOpen(state),
        editMenuOpen: editMenuOpen(state),
        connectMenuOpen: connectMenuOpen(state),
        HackMenuOpen: HackMenuOpen(state),
        isUpdating: getIsUpdating(loadingState),
        isRealtimeMode: state.scratchGui.programMode.isRealtimeMode,
        isRtl: state.locales.isRtl,
        messagesByLocale: state.locales.messagesByLocale,
        isShowingProject: getIsShowingProject(loadingState),
        isSupportSwitchMode: state.scratchGui.programMode.isSupportSwitchMode,
        isToolboxUpdating: state.scratchGui.toolbox.isToolboxUpdating,
        languageMenuOpen: languageMenuOpen(state),
        locale: state.locales.locale,
        loginMenuOpen: loginMenuOpen(state),
        projectTitle: state.scratchGui.projectTitle,
        projectRunning: state.scratchGui.vmStatus.running,
        realtimeConnection: state.scratchGui.connectionModal.realtimeConnection,
        sessionExists: state.session && typeof state.session.session !== 'undefined',
        username: user ? user.username : null,
        userOwnsProject: ownProps.authorUsername && user &&
            (ownProps.authorUsername === user.username),
        stageSizeMode: state.scratchGui.stageSize.stageSize,
        vm: state.scratchGui.vm,
        peripheralName: state.scratchGui.connectionModal.peripheralName,
        deviceId: state.scratchGui.device.deviceId,
        deviceName: state.scratchGui.device.deviceName,
        allFiles: state.scratchGui.filesData.allFiles,
    };
};

const mapDispatchToProps = dispatch => ({
    autoUpdateProject: () => dispatch(autoUpdateProject()),
    onOpenTipLibrary: () => dispatch(openTipsLibrary()),
    onClickAccount: () => dispatch(openAccountMenu()),
    onRequestCloseAccount: () => dispatch(closeAccountMenu()),
    onClickFile: () => dispatch(openFileMenu()),
    onRequestCloseFile: () => dispatch(closeFileMenu()),
    onClickSetting: () => dispatch(openSettingMenu()),
    onRequestCloseSetting: () => dispatch(closeSettingMenu()),
    onSelectLocale: locale => {
        dispatch(selectLocale(locale));
        dispatch(closeLanguageMenu());
    },
    onClickEdit: () => dispatch(openEditMenu()),
    onClickHack: () => dispatch(openHackMenu()),
    onRequestCloseHack: () => dispatch(closeHackMenu()),
    onRequestCloseEdit: () => dispatch(closeEditMenu()),
    onClickConnect: () => dispatch(openConnectMenu()),
    onRequestCloseConnect: () => dispatch(closeConnectMenu()),
    onClickLanguage: () => dispatch(openLanguageMenu()),
    onRequestCloseLanguage: () => dispatch(closeLanguageMenu()),
    onClickLogin: () => dispatch(openLoginMenu()),
    onRequestCloseLogin: () => dispatch(closeLoginMenu()),
    onRequestOpenAbout: () => dispatch(openAboutMenu()),
    onRequestCloseAbout: () => dispatch(closeAboutMenu()),
    onClickNew: needSave => dispatch(requestNewProject(needSave)),
    onClickRemix: () => dispatch(remixProject()),
    onClickSave: () => dispatch(manualUpdateProject()),
    onClickSaveAsCopy: () => dispatch(saveProjectAsCopy()),
    onSeeCommunity: () => dispatch(setPlayer(true)),
    onSetUploadMode: () => {
        dispatch(setUploadMode());
        dispatch(setRealtimeConnection(false));
    },
    onOpenMLforKidsLibrary: () => dispatch(openMLForKidsLibrary()),
    onOpenDemoLibrary: () => dispatch(openDemoLibrary()),
    onOpenTestLibrary: () => dispatch(openTestLibrary()),
    onSetRealtimeConnection: state => dispatch(setRealtimeConnection(state)),
    onSetRealtimeMode: () => dispatch(setRealtimeMode()),
    onSetStageLarge: () => dispatch(setStageSize(STAGE_SIZE_MODES.large)),
    onOpenConnectionModal: () => dispatch(openConnectionModal()),
    onOpenUploadProgress: () => dispatch(openUploadProgress()),
    onDisconnect: () => {
        dispatch(clearConnectionModalPeripheralName());
        dispatch(setRealtimeConnection(false));
    },
    onSetUpdate: message => {
        dispatch(setUpdate(message));
        dispatch(openUpdateModal());
    },
    onNoPeripheralIsConnected: () => showAlertWithTimeout(dispatch, 'connectAPeripheralFirst'),
    onWorkspaceIsEmpty: () => showAlertWithTimeout(dispatch, 'workspaceIsEmpty'),
    onWorkspaceIsNotEmpty: () => showAlertWithTimeout(dispatch, 'workspaceIsNotEmpty'),

});
